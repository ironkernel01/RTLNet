# RTLNet Nexys Video CLI command set: the input of cli/gen_menu.py (format in
# cli/MenuBuilder_hierarchical.md). Lines starting with '#' are comments.
#
#   REVISION:<MM-DD-YYYY>:<change>   (revision history of the command set)
#   MODE-ROOT:<mode>:<prompt>:<mode description>
#   MODE-ENTER:<FUNCTION or ->:<from mode>:<COMMAND>:<to mode>:<prompt>:<mode description>:<DESCRIPTION>
#   KEYWORD:<mode>:<KEYWORD PATH>:<DESCRIPTION>
#   <FUNCTION>:<mode>:<COMMAND>:<DESCRIPTION>
#
# Modes: exec, priv, config, config-if, config-dhcp, config-sock. Commands of
# exec are also available in priv. Built-ins (engine): EXIT, HELP, ?, !! in
# every mode; END in priv and the configuration modes.

# ---------------------------------------------------------------------------
# Modes
# ---------------------------------------------------------------------------
MODE-ROOT:exec:rtlnet>:User exec
MODE-ENTER:-:exec:ENABLE:priv:rtlnet#:Privileged exec:Enter privileged exec mode
MODE-ENTER:-:priv:CONFIGURE TERMINAL:config:rtlnet(config)#:Global configuration:Enter global configuration mode
MODE-ENTER:cli_enter_interface:config:INTERFACE [IFNAME]:config-if:rtlnet(config-if)#:Interface configuration:Configure an interface (eth0)
MODE-ENTER:-:config:IP DHCP:config-dhcp:rtlnet(config-dhcp)#:DHCP client configuration:Configure the DHCP client
MODE-ENTER:cli_enter_socket:config:SOCKET [INDEX]:config-sock:rtlnet(config-sock)#:UDP socket configuration:Configure a UDP socket (index 0..3)

# ---------------------------------------------------------------------------
# Keyword groups (help text of a keyword shared by several commands)
# ---------------------------------------------------------------------------
KEYWORD:exec:SHOW:Show running system information
KEYWORD:priv:DEBUG:Raw register access
KEYWORD:priv:DDR:DDR3 memory
KEYWORD:priv:CAMERA:Video source: mode, source switch, frame rate
KEYWORD:priv:FRAMEBUFFER:Video frame buffer in DDR3
KEYWORD:priv:STREAM:H.264 MPEG-TS stream over UDP
KEYWORD:priv:ENCODER:H.264 encoder settings
KEYWORD:priv:LED:Set a user LED (2..7)
KEYWORD:priv:PHY:PHY registers over MDIO
KEYWORD:priv:PHY MMD:MMD registers, Clause 45 over Clause 22 (registers 13/14)
KEYWORD:priv:PHY C45:Native Clause 45 frames (the RTL8211E answers only phy mmd)
KEYWORD:config:IP:Global IP configuration
KEYWORD:config:NO:Negate a command or set its default
KEYWORD:config:NO ECHO:UDP echo server
KEYWORD:config:ECHO:UDP echo server
KEYWORD:config:TCP:TCP applications on lwIP (CPU network path)
KEYWORD:config:IPERF:iperf servers (UDP in RTL, TCP in firmware)
KEYWORD:config:IPERF TCP:TCP iperf server (lwIP lwiperf, default port 5001)
KEYWORD:config:TCP ECHO:TCP echo server (lwIP, default port 7)
KEYWORD:config:OFFLOAD:Checksum offload on the CPU path (hardware IPv4/TCP/UDP checksums)
KEYWORD:config:CLASSIFIER:CPU-path classifier (filter tables, policers, mode)
KEYWORD:config:NO CLASSIFIER:Clear a classifier entry or disable a policer
KEYWORD:config-if:STORM-CONTROL:Broadcast storm limiter (classifier policer 0)
KEYWORD:config-if:STORM-CONTROL BROADCAST:Broadcast frames to the CPU: rate limit, enable, disable
KEYWORD:config-if:NO STORM-CONTROL:Disable the broadcast storm limiter
KEYWORD:config-if:IP:Interface IP configuration
KEYWORD:config-if:NO:Negate a command or set its default
KEYWORD:config-if:NO IP:Interface IP configuration
KEYWORD:config-dhcp:NO:Negate a command or set its default
KEYWORD:config-sock:NO:Negate a command or set its default

# ---------------------------------------------------------------------------
# User exec (also privileged exec)
# ---------------------------------------------------------------------------
cli_show_version:exec:SHOW VERSION:Board, register map, build and firmware identity
cli_show_system:exec:SHOW SYSTEM:Clocks, uptime, switches, buttons and PHY pins
cli_show_interfaces:exec:SHOW INTERFACES:Ethernet link, MAC status and frame counters
cli_show_phy:exec:SHOW PHY:PHY identity, RX delay, link, advertisement and partner
cli_show_ip:exec:SHOW IP:Node MAC, static and effective IPv4 address, mask and gateway
cli_show_arp:exec:SHOW ARP:ARP responder counters
cli_show_dhcp:exec:SHOW DHCP:DHCP client state, lease and counters
cli_show_iperf:exec:SHOW IPERF:iperf servers: UDP (RTL core) and TCP (firmware, lwIP)
cli_show_sockets:exec:SHOW SOCKETS:UDP socket table, UDP echo and counters
cli_show_counters:exec:SHOW COUNTERS:Frame counters of every core
cli_show_tcp:exec:SHOW TCP:lwIP interface, TCP echo server, connections and counters
cli_show_cpu_path:exec:SHOW CPU-PATH:CPU network path: DMA frames and errors, wrapper CPU-path counters
cli_show_storm:exec:SHOW STORM-CONTROL:Broadcast storm limiter: policer 0 and its counters
cli_show_classifier:exec:SHOW CLASSIFIER:CPU-path classifier: mode, tables, ACL, policers, counters
cli_show_offload:exec:SHOW OFFLOAD:Checksum offload: discovery, CTRL, lwIP flags, counters, zero-copy options
cli_show_ddr:exec:SHOW DDR:DDR3 calibration, caches, arbiter identity, error status and counters
cli_show_camera:exec:SHOW CAMERA:Video source: mode, source switch, active source, frame rate and count
cli_show_framebuffer:exec:SHOW FRAMEBUFFER:Video frame buffer: slots, latest published frame, reference counts, write DMA
cli_show_stream:exec:SHOW STREAM:H.264 stream: pictures, bytes, bit rate, dropped pictures, TS packets and mux CPU load
cli_show_encoder:exec:SHOW ENCODER:H.264 encoder: QP, mode, status and per-engine cycle statistics

# ---------------------------------------------------------------------------
# Privileged exec
# ---------------------------------------------------------------------------
cli_show_running:priv:SHOW RUNNING-CONFIG:Current configuration
cli_disable:priv:DISABLE:Leave privileged exec mode
cli_ddr_test:priv:DDR TEST {SIZE}:DDR3 memory test from 0x80000000: data bus, address bus, random fill and verify, with MB/s (SIZE in MiB, 1..512, default 16)
cli_camera_mode:priv:CAMERA MODE [MODE]:Video mode: 720p30 or 1080p30 (effective at the next frame)
cli_camera_source:priv:CAMERA SOURCE [SOURCE]:Video source: auto, camera or pattern
cli_camera_rate:priv:CAMERA RATE [FPS]:Frame rate in frames per second, 1..30 (30 is the reset value)
cli_framebuffer_crc:priv:FRAMEBUFFER CRC:CRC-32 of the latest published frame, read through the cache
cli_stream_h264:priv:STREAM H264 [IP] [PORT]:Start the H.264 MPEG-TS stream over UDP to a host (VLC: udp://@:PORT)
cli_stream_stop:priv:STREAM STOP:Stop the H.264 stream
cli_encoder_qp:priv:ENCODER QP [QP]:H.264 quantizer 0..51, fixed (the driver raises values below 12 to 12; default 28)
cli_encoder_gop:priv:ENCODER GOP [GOP]:IDR picture every GOP pictures, P pictures between, 1..255 (1 = all IDR; default 30)
cli_debug_read:priv:DEBUG READ [ADDR]:Read a 32-bit word (hex 0x... or decimal address)
cli_debug_write:priv:DEBUG WRITE [ADDR] [VALUE]:Write a 32-bit word
cli_phy_read:priv:PHY READ [REG]:Read a PHY register (0..31) over MDIO
cli_phy_write:priv:PHY WRITE [REG] [VALUE]:Write a PHY register (0..31) over MDIO
cli_phy_dump:priv:PHY DUMP:Read PHY registers 0..31
cli_phy_mmd_read:priv:PHY MMD READ [DEVAD] [REG]:Read an MMD register (Clause 45 over Clause 22)
cli_phy_mmd_write:priv:PHY MMD WRITE [DEVAD] [REG] [VALUE]:Write an MMD register (Clause 45 over Clause 22)
cli_phy_c45_read:priv:PHY C45 READ [DEVAD] [REG]:Native Clause 45 read (the RTL8211E answers only phy mmd)
cli_phy_c45_write:priv:PHY C45 WRITE [DEVAD] [REG] [VALUE]:Native Clause 45 write (the RTL8211E answers only phy mmd)
cli_phy_reset:priv:PHY RESET:Hardware PHY reset, then InitPHY (RX delay, LEDs, EEE, advertisement)
cli_clear_counters:priv:CLEAR COUNTERS:Clear the counters of every core
cli_led_on:priv:LED [N] ON:Turn user LED n (2..7) on
cli_led_off:priv:LED [N] OFF:Turn user LED n (2..7) off
cli_reload:priv:RELOAD:Restart the firmware (soft restart)

# ---------------------------------------------------------------------------
# Global configuration
# ---------------------------------------------------------------------------
cli_ip_gateway:config:IP DEFAULT-GATEWAY [ADDR]:Set the static IPv4 default gateway
cli_no_ip_gateway:config:NO IP DEFAULT-GATEWAY:Remove the static default gateway (0.0.0.0)
cli_echo_enable:config:ECHO ENABLE {SOCKET}:Enable the UDP echo server (configured socket, or the one given)
cli_no_echo_enable:config:NO ECHO ENABLE:Disable the UDP echo server
cli_echo_socket:config:ECHO SOCKET [INDEX]:Echo on socket 0..3, or "any"
cli_iperf_port:config:IPERF PORT [PORT]:iperf server UDP port (default 5001)
cli_iperf_tcp_enable:config:IPERF TCP ENABLE:Enable the TCP iperf server
cli_iperf_tcp_disable:config:IPERF TCP DISABLE:Disable the TCP iperf server
cli_iperf_tcp_port:config:IPERF TCP PORT [PORT]:TCP iperf server port (default 5001)
cli_tcp_echo_enable:config:TCP ECHO ENABLE:Enable the TCP echo server
cli_tcp_echo_disable:config:TCP ECHO DISABLE:Disable the TCP echo server (open connections stay)
cli_tcp_echo_port:config:TCP ECHO PORT [PORT]:TCP echo server port (default 7)
cli_offload:config:OFFLOAD [WHAT] [STATE]:Checksum offload: tx-ip, tx-l4, rx or all, on or off (as far as the build has it; lwIP follows)
cli_cls_mode:config:CLASSIFIER MODE [MODE]:Mode: drop-all, pass-all, pass-on-match or drop-on-match
cli_cls_policer:config:CLASSIFIER POLICER [INDEX] [PKTS] [MS] {DISABLE}:Policer 0..3: frames per window in ms (or disable)
cli_cls_mac:config:CLASSIFIER MAC [ENTRY] [WHICH] [MAC] {W1} {W2} {W3} {W4}:MAC entry (0..15): dst|src <mac> gate|classify|police <p>
cli_cls_ethertype:config:CLASSIFIER ETHERTYPE [ENTRY] [TYPE] {W1} {W2} {W3} {W4}:EtherType entry (0..7): <type> [to <type>] gate|classify|police <p>
cli_cls_acl:config:CLASSIFIER ACL [ENTRY] [PROTO] {W1} {W2} {W3} {W4} {W5} {W6} {W7} {W8} {W9} {W10}:ACL entry: tcp|udp|any src <ip>/<len> dst <ip>/<len> [sport <lo>[-<hi>]] [dport <lo>[-<hi>]] gate|classify|police <p>
cli_cls_defaults:config:CLASSIFIER DEFAULTS:Broadcast storm default (pass-all, broadcast 10 per 100 ms)
cli_no_cls_mac:config:NO CLASSIFIER MAC [ENTRY]:Clear a MAC entry
cli_no_cls_ethertype:config:NO CLASSIFIER ETHERTYPE [ENTRY]:Clear an EtherType entry
cli_no_cls_acl:config:NO CLASSIFIER ACL [ENTRY]:Clear an ACL entry
cli_no_cls_policer:config:NO CLASSIFIER POLICER [INDEX]:Disable a policer

# ---------------------------------------------------------------------------
# Interface configuration (eth0)
# ---------------------------------------------------------------------------
cli_if_speed:config-if:SPEED [MBPS]:Link speed: 10, 100, 1000 or auto
cli_if_duplex:config-if:DUPLEX [MODE]:Duplex: full, half or auto (the MAC is full duplex only)
cli_if_autoneg:config-if:AUTONEG:Enable auto-negotiation (speed/duplex limit the advertisement)
cli_if_no_autoneg:config-if:NO AUTONEG:Disable auto-negotiation (forced 10 or 100 Mb/s)
cli_if_advertise:config-if:ADVERTISE [MODES]:Advertise modes, e.g. 10f,100f,1000f or all
cli_if_no_advertise:config-if:NO ADVERTISE:Advertisement from speed and duplex again
cli_if_shutdown:config-if:SHUTDOWN:Disable the interface (MAC receive and transmit)
cli_if_no_shutdown:config-if:NO SHUTDOWN:Enable the interface
cli_if_ip_address:config-if:IP ADDRESS [ADDR] [MASK]:Set the static IPv4 address and mask
cli_if_no_ip_address:config-if:NO IP ADDRESS:Remove the static IPv4 address (0.0.0.0)
cli_if_ip_netmask:config-if:IP NETMASK [MASK]:Set the static netmask
cli_if_mac:config-if:MAC-ADDRESS [MAC]:Set the node MAC address (aa:bb:cc:dd:ee:ff)
cli_if_storm:config-if:STORM-CONTROL BROADCAST [PKTS] [MS]:Pass at most PKTS broadcast frames per MS milliseconds to the CPU
cli_if_storm_enable:config-if:STORM-CONTROL BROADCAST ENABLE:Enable the broadcast storm limiter
cli_if_storm_disable:config-if:STORM-CONTROL BROADCAST DISABLE:Disable the broadcast storm limiter (every broadcast passes)
cli_if_no_storm:config-if:NO STORM-CONTROL BROADCAST:Disable the broadcast storm limiter

# ---------------------------------------------------------------------------
# DHCP client configuration
# ---------------------------------------------------------------------------
cli_dhcp_start:config-dhcp:START:Start DHCP discovery now
cli_dhcp_stop:config-dhcp:STOP:Stop the DHCP client and release the lease
cli_dhcp_renew:config-dhcp:RENEW:Renew the lease now
cli_dhcp_auto:config-dhcp:AUTO:Start DHCP automatically on link up
cli_dhcp_no_auto:config-dhcp:NO AUTO:Do not start DHCP on link up

# ---------------------------------------------------------------------------
# UDP socket configuration
# ---------------------------------------------------------------------------
cli_sock_open:config-sock:OPEN [PORT]:Open the socket on a local UDP port
cli_sock_close:config-sock:CLOSE:Close the socket
cli_sock_remote:config-sock:REMOTE [ADDR] [PORT] {MAC}:Remote IPv4 address, UDP port and next-hop MAC
cli_sock_connect:config-sock:CONNECT-ON-RECEIVE:Follow the sender of the last datagram
cli_sock_no_connect:config-sock:NO CONNECT-ON-RECEIVE:Keep the configured remote address
