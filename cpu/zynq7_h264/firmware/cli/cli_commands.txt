# Purpose: reduced ARM camera/encoder CLI; Overview: MenuBuilder source of truth.
# Revision: 11-14-2026.
REVISION:11-14-2026:Initial Zybo camera demonstration commands
REVISION:11-15-2026:Own camera receiver commands (SHOW RX, RX ...)
MODE-ROOT:exec:rtlnet>:User exec
MODE-ENTER:-:exec:ENABLE:priv:rtlnet#:Privileged exec:Enter privileged exec
KEYWORD:exec:SHOW:Show current status
KEYWORD:priv:ENCODER:Configure encoder
KEYWORD:priv:STREAM:Control MPEG-TS UDP stream
KEYWORD:priv:RX:Configure the own camera receiver
cli_version:exec:SHOW VERSION:Firmware and hardware identity
cli_ip:exec:SHOW IP:DHCP address and PHY identity
cli_camera:exec:SHOW CAMERA:Camera mode and capture counters
cli_rx:exec:SHOW RX:Own receiver state and all counters
cli_encoder:exec:SHOW ENCODER:Encoder picture and error statistics
cli_stream:exec:SHOW STREAM:MPEG-TS and UDP statistics
cli_qp:priv:ENCODER QP [VALUE]:Constant QP 0..51
cli_gop:priv:ENCODER GOP [VALUE]:IDR period 1..255
cli_bitrate:priv:ENCODER BITRATE [KBPS]:Target kbps 0..100000; zero selects constant QP
cli_start:priv:STREAM START [IP] [PORT]:Start H264 MPEG-TS UDP to IPv4 destination
cli_stop:priv:STREAM STOP:Stop stream and release capture slot
cli_mode:priv:CAMERA MODE [MODE]:720p30 or 1080p15; requires stream stopped
cli_rx_clear:priv:RX CLEAR:Clear the receiver counters
cli_rx_taps:priv:RX TAPS [LANE0] [LANE1]:Input delay taps 0..31 of the two data lanes (78 ps each)
cli_rx_wb:priv:RX WB [R] [G] [B]:White balance gains, 256 = 1.0, 0..4095
cli_rx_black:priv:RX BLACK [LEVEL]:Black level in 10-bit units 0..1023
cli_rx_phase:priv:RX PHASE [VALUE]:Bayer phase 0..3 of the first pixel
cli_rx_gamma:priv:RX GAMMA [ONOFF]:Gamma curve 0 or 1
cli_rx_geq:priv:RX GEQ [ONOFF]:Green equalisation 0 or 1
cli_rx_bitrev:priv:RX BITREV [ONOFF]:Deserializer bit order reversal 0 or 1
cli_rx_settle:priv:RX SETTLE [CLOCKS]:HS settle time in 200 MHz clocks 0..255
